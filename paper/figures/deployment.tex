% Practical deployment-architecture diagram (TikZ).
% Used in main.tex as:  \input{figures/deployment}
% Sized to fit a two-column IEEE figure* (max ~17.6cm).

\begin{tikzpicture}[
  font=\scriptsize,
  align=center,
  every path/.style={-{Latex[length=2.2mm]}, thick, line width=0.6pt},
  src/.style={rectangle, rounded corners=2pt, draw=black, thick,
              fill=yellow!15, minimum height=6mm,
              text width=1.7cm, inner sep=2pt},
  proc/.style={rectangle, rounded corners=2pt, draw=black, thick,
               minimum height=10mm, text width=2.5cm, inner sep=3pt},
  bus/.style={proc, fill=green!15},
  hmi/.style={proc, fill=orange!18, text width=2.4cm},
]

\node[src]                          (grid) {AC grid};
\node[src, below=1.5mm of grid]     (pv)   {Solar PV};
\node[src, below=1.5mm of pv]       (dg)   {DG / EV};

\node[proc, fill=blue!10, right=6mm of pv] (sense)
  {\textbf{Sensing}\\
   {\tiny ZMPT101B (V)}\\
   {\tiny ACS712 (I)}\\
   {\tiny ADC @ 5\,kHz}};

\node[proc, fill=blue!22, right=6mm of sense] (edge)
  {\textbf{Edge node}\\
   {\tiny Raspberry Pi /}\\
   {\tiny ESP32 / ARM}\\
   {\tiny Stage~1 + Stage~2}};

\node[bus, right=6mm of edge] (bus)
  {\textbf{Comms}\\
   {\tiny RS485 / Modbus}\\
   {\tiny or SocketIO}};

\node[hmi, right=6mm of bus, yshift=6mm]  (scada) {\textbf{SCADA HMI}};
\node[hmi, right=6mm of bus, yshift=-6mm] (dash)  {\textbf{Browser dashboard}};

\draw (grid.east) -- (sense.west);
\draw (pv.east)   -- (sense.west);
\draw (dg.east)   -- (sense.west);
\draw (sense)     -- (edge);
\draw (edge)      -- (bus);
\draw (bus.east)  -- (scada.west);
\draw (bus.east)  -- (dash.west);

\end{tikzpicture}
